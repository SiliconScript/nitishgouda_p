2FF Synchronizer

Eight parallel instances handle each bit of pixel_data, plus dedicated synchronizers for valid strobes. Prevents metastability crossing into clk_core.

sync_2ff.v
module sync_2ff(
  input wire clk, rst_n,
  input wire async_in,
  output reg sync_out
);
  reg ff1;
  always @(posedge clk or negedge rst_n)
    if (!rst_n) {ff1,sync_out} <= 2'b0;
    else {sync_out,ff1} <= {ff1,async_in};
endmodule
